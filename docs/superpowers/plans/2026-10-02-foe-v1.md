# Foe v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A RuneLite overlay showing the live HP, combat levels and elemental weakness of the monster you are fighting, using client data only.

**Architecture:** Pure, unit-tested logic (`TargetTracker`, `HpEstimate`, `SnapshotFactory`, `WeaknessDecoder`) turns RuneLite events into one immutable `TargetSnapshot` per game tick. `FoeOverlay` only draws that snapshot. `FoePlugin` is the wiring.

**Tech Stack:** Java 11, Gradle 8.10, RuneLite client API (`latest.release`), Lombok 1.18.30, JUnit 4.12.

**Spec:** `docs/superpowers/specs/2026-10-02-foe-design.md`. Read its 2026-10-02 addendum too, because it amends target selection.

**Branch:** all work lands on feature branches and goes through a PR. The pre-push gate blocks pushes to `main`.

**API facts this plan relies on** were checked against RuneLite `master` `d8e7d1e` (2026-09-30) `[measured]`: `NPCComposition.getStats()` and the `STAT_*` constants; `NPCManager.getHealth(int)`; `Actor.getHealthRatio/getHealthScale/getInteracting`; `VarPlayerID.LAST_NPC_ELEMENTAL_WEAKNESS = 5536`; `VarbitChanged.getVarpId/getValue`; `InteractingChanged.getSource/getTarget`; `HitsplatApplied.getActor/getHitsplat`; `Hitsplat.isMine()`; `ActorDeath.getActor`; `NpcDespawned.getNpc`; `Client.getTopLevelWorldView().npcs()` (iterable); `BackgroundComponent(Color, Rectangle)`; `Overlay.setPosition`; `Units.SECONDS/PERCENT`; `Text.removeTags`; and `OpponentInfoOverlay.java:175-198` (the HP formula ported in Task 3).

```java
import lombok.extern.slf4j.Slf4j;
import net.runelite.api.Actor;
import net.runelite.api.Client;
import net.runelite.api.NPC;

import net.runelite.api.events.VarbitChanged;
import net.runelite.api.gameval.VarPlayerID;
import net.runelite.client.eventbus.Subscribe;
import net.runelite.client.plugins.Plugin;
import net.runelite.client.plugins.PluginDescriptor;

@Slf4j
@PluginDescriptor(
	name = "Foe",
	description = "Live HP, combat levels and elemental weakness of the monster you're fighting",
	tags = {"target", "monster", "npc", "weakness", "opponent", "hp", "combat"}
)
public class FoePlugin extends Plugin
{
	@Inject
	private Client client;

	// PROBE (Task 1) — removed in Task 7. Logs varp 5536 so its encoding can be decoded.
	@Subscribe
	public void onVarbitChanged(VarbitChanged e)
	{
		if (e.getVarpId() != VarPlayerID.LAST_NPC_ELEMENTAL_WEAKNESS)
		{
			return;
		}
		long v = e.getValue();
		log.info("[foe-probe] varp5536 changed: dec={} hex={} bin={} | {}",
			v, Long.toHexString(v), Long.toBinaryString(v), describe(client.getLocalPlayer().getInteracting()));
	}

	@Subscribe
	public void onInteractingChanged(InteractingChanged e)
	{
		if (e.getSource() != client.getLocalPlayer())
		{
			return;
		}
		log.info("[foe-probe] player now targets {} | varp5536 currently {}",
			describe(e.getTarget()), client.getVarpValue(VarPlayerID.LAST_NPC_ELEMENTAL_WEAKNESS));
	}

	private static String describe(Actor a)
	{
		if (!(a instanceof NPC))
		{
			return "no NPC";
		}
		NPC n = (NPC) a;
		return n.getName() + " (id " + n.getId() + ", idx " + n.getIndex() + ")";
	}
}
```

```java
package com.foe;

/**
 * Decides which NPC is the foe. Pure: no RuneLite types, so time and NPC indices are passed in.
 * Sticky by design (spec addendum 1): in multi-combat, another NPC's hit does not steal a live target.
 */
final class TargetTracker
{
	static final int NONE = -1;

	private int target = NONE;
	private long lastActiveMs;

	/** The player attacked this NPC. Always wins. */
	void playerAttacks(int npcIndex, long nowMs)
	{
		target = npcIndex;
		lastActiveMs = nowMs;
	}

	/** An NPC hit the player. Adopted only when there is no live target. */
	void hitBy(int npcIndex, long nowMs, long lingerMs)
	{
		if (npcIndex == target)
		{
			lastActiveMs = nowMs;
			return;
		}
		if (current(nowMs, lingerMs) == NONE)
		{
			target = npcIndex;
			lastActiveMs = nowMs;
		}
	}

	/** Combat with this NPC is still going on (either side is interacting with the other). */
	void stillFighting(int npcIndex, long nowMs)
	{
		if (npcIndex == target)
		{
			lastActiveMs = nowMs;
		}
	}

	/** The NPC died or despawned. Its index may be reused, so it must not linger. */
	void gone(int npcIndex)
	{
		if (npcIndex == target)
		{
			target = NONE;
		}
	}

	int current(long nowMs, long lingerMs)
	{
		if (target != NONE && nowMs - lastActiveMs > lingerMs)
		{
			target = NONE;
		}
		return target;
	}
}
```

```java
package com.foe;

/**
 * Estimated HP from the health bar. The game sends only a fraction (ratio/scale), so this returns
 * the midpoint of the possible range. Ported from RuneLite OpponentInfoOverlay (d8e7d1e, :175-198).
 */
final class HpEstimate
{
	static final int UNKNOWN = -1;

	private HpEstimate()
	{
	}

	static int estimate(int ratio, int scale, int maxHp)
	{
		if (ratio < 0 || scale <= 0 || maxHp <= 0)
		{
			return UNKNOWN;
		}
		if (ratio == 0)
		{
			return 0;
		}
		int min = 1;
		int max;
		if (scale > 1)
		{
			if (ratio > 1)
			{
				// ratio == 1 is excluded: the server forces ratio 0 at 0 HP, so ratio 1 only bounds the top
				min = (maxHp * (ratio - 1) + scale - 2) / (scale - 1);
			}
			max = Math.min((maxHp * ratio - 1) / (scale - 1), maxHp);
		}
		else
		{
			max = maxHp;
		}
		return (min + max + 1) / 2;
	}
}
```

```java
package com.foe;

import net.runelite.api.NPCComposition;
import net.runelite.client.util.Text;

/** Raw client values -> TargetSnapshot. Pure apart from Text.removeTags, so it is unit-testable. */
final class SnapshotFactory
{
	private SnapshotFactory()
	{
	}

	/**
	 * @param stats          NPCComposition.getStats(); may be null or short
	 * @param fallbackMaxHp  NPCManager.getHealth(id); may be null
	 * @return null when there is nothing nameable to show
	 */
	static TargetSnapshot build(String rawName, int combatLevel, int[] stats, int ratio, int scale,
		Integer fallbackMaxHp, Weakness weakness)
	{
		if (rawName == null)
		{
			return null;
		}
		String name = Text.removeTags(rawName).trim();
		if (name.isEmpty() || name.equals("null"))
		{
			return null;
		}
		int maxHp = stat(stats, NPCComposition.STAT_HITPOINTS);
		if (maxHp <= 0 && fallbackMaxHp != null)
		{
			maxHp = fallbackMaxHp;
		}
		return new TargetSnapshot(name, combatLevel, HpEstimate.estimate(ratio, scale, maxHp), Math.max(maxHp, 0),
			ratio, scale,
			stat(stats, NPCComposition.STAT_ATTACK), stat(stats, NPCComposition.STAT_STRENGTH),
			stat(stats, NPCComposition.STAT_DEFENCE), stat(stats, NPCComposition.STAT_RANGED),
			stat(stats, NPCComposition.STAT_MAGIC), weakness);
	}

	private static int stat(int[] stats, int i)
	{
		return stats != null && i < stats.length ? Math.max(stats[i], 0) : 0;
	}
}
```

```java
package com.foe;

import java.awt.Color;
import java.awt.Dimension;
import java.awt.FontMetrics;
import java.awt.Graphics2D;
import java.awt.Rectangle;
import java.util.ArrayList;
import java.util.List;
import javax.inject.Inject;
import net.runelite.client.ui.overlay.Overlay;
import net.runelite.client.ui.overlay.OverlayPosition;
import net.runelite.client.ui.overlay.components.BackgroundComponent;
import net.runelite.client.ui.overlay.components.ComponentConstants;

class FoeOverlay extends Overlay
{
	private static final int PAD = 4;
	private static final int GAP = 8;
	private static final int BAR_W = 80;
	private static final int BAR_H = 12;
	private static final Color BAR_FG = new Color(82, 161, 82);
	private static final Color BAR_BG = new Color(255, 255, 255, 60);
	private static final Color DIVIDER = new Color(255, 255, 255, 70);
	private static final Color DIM = new Color(255, 255, 255, 150);

	private final FoePlugin plugin;
	private final FoeConfig config;

	@Inject
	FoeOverlay(FoePlugin plugin, FoeConfig config)
	{
		this.plugin = plugin;
		this.config = config;
		setPosition(OverlayPosition.TOP_LEFT);
	}

	@Override
	public Dimension render(Graphics2D g)
	{
		TargetSnapshot s = plugin.getSnapshot();
		if (s == null)
		{
			return null;
		}
		FontMetrics fm = g.getFontMetrics();
		List<String> texts = segments(s);
		boolean bar = s.getHpScale() > 0 && s.getHpRatio() >= 0 && s.getHpRatio() <= s.getHpScale();

		// Layout: [name] | [bar] [hp text] | [levels] | [weakness]
		int w = PAD;
		for (String t : texts)
		{
			w += fm.stringWidth(t) + GAP;
		}
		if (bar)
		{
			w += BAR_W + GAP;
		}
		w = w - GAP + PAD;
		int h = Math.max(fm.getHeight(), BAR_H) + PAD * 2;

		int alpha = Math.round(config.backgroundOpacity() * 255 / 100f);
		if (alpha > 0)
		{
			Color base = ComponentConstants.STANDARD_BACKGROUND_COLOR;
			new BackgroundComponent(new Color(base.getRed(), base.getGreen(), base.getBlue(), alpha),
				new Rectangle(0, 0, w, h)).render(g);
		}

		int x = PAD;
		int baseline = PAD + (h - PAD * 2 + fm.getAscent() - fm.getDescent()) / 2;
		for (int i = 0; i < texts.size(); i++)
		{
			String t = texts.get(i);
			if (i > 0)
			{
				g.setColor(DIVIDER);
				g.drawLine(x - GAP / 2, PAD, x - GAP / 2, h - PAD);
			}
			g.setColor(i == 0 ? Color.WHITE : DIM);
			g.drawString(t, x, baseline);
			x += fm.stringWidth(t) + GAP;
			// The bar always follows the name (texts[0], never empty). It is anchored to the name
			// rather than to the HP text's slot, because the HP text is omitted in bar-only mode.
			if (i == 0 && bar)
			{
				int y = (h - BAR_H) / 2;
				g.setColor(BAR_BG);
				g.fillRect(x, y, BAR_W, BAR_H);
				g.setColor(BAR_FG);
				g.fillRect(x, y, Math.round(BAR_W * (float) s.getHpRatio() / s.getHpScale()), BAR_H);
				x += BAR_W + GAP;
			}
		}
		return new Dimension(w, h);
	}

	/** Text segments in order: name, hp text, levels, weakness. Empty = omitted (no placeholders, per spec). */
	private List<String> segments(TargetSnapshot s)
	{
		List<String> out = new ArrayList<>();
		out.add(config.showCombatLevel() && s.getCombatLevel() > 0 ? s.getName() + "  " + s.getCombatLevel() : s.getName());
		out.add(hpText(s));
		out.add(config.detail() == FoeConfig.Detail.FULL ? levels(s) : "");
		Weakness wk = s.getWeakness();
		out.add(config.showWeakness() && wk != null ? cap(wk.getElement().name()) : "");
		out.removeIf(String::isEmpty);
		return out;
	}

	private String hpText(TargetSnapshot s)
	{
		if (s.getMaxHp() <= 0 || s.getHp() < 0)
		{
			return ""; // spec: unknown max HP -> bar only
		}
		switch (config.hpDisplay())
		{
			case NUMBER:
				return "~" + s.getHp() + "/" + s.getMaxHp();
			case PERCENT:
				return Math.round(100f * s.getHp() / s.getMaxHp()) + "%";
			default:
				return "";
		}
	}

	private String levels(TargetSnapshot s)
	{
		StringBuilder b = new StringBuilder();
		level(b, "Att", s.getAttack());
		level(b, "Str", s.getStrength());
		level(b, "Def", s.getDefence());
		level(b, "Rng", s.getRanged());
		level(b, "Mag", s.getMagic());
		return b.toString().trim();
	}

	private void level(StringBuilder b, String label, int v)
	{
		int floor = config.hideIrrelevantLevels() ? 1 : 0;
		if (v > floor)
		{
			b.append(label).append(' ').append(v).append("  ");
		}
	}

	private static String cap(String s)
	{
		return s.charAt(0) + s.substring(1).toLowerCase();
	}
}
```

### Task 7: FoePlugin wiring

> **Amendments from the Task 2 review (2026-10-03). These override the code below where they conflict.**
> 1. **A player's landed hit (`isMine`) calls `tracker.playerHit(idx, now, linger)`, not `playerAttacks`.** A hit already in flight can land after a target switch (`docs/probe/raw.log` lines 265–267). `playerAttacks` is for engagement only (`onInteractingChanged` to an NPC). The tests that pin this are in `TargetTrackerTest`.
> 2. **Take one `now` per handler,** and pass it to both the refresh and the `current()` read. With linger 0 the live check is `<=` to the millisecond, so two separate `now()` calls can blank an actively fought target.
> 3. **Use a monotonic clock:** `System.nanoTime() / 1_000_000`, not `System.currentTimeMillis()`. See the `TargetTracker` Javadoc.
> 4. **`hitBy` must name the NPC that actually hit.** The code below picks the first NPC interacting with the player, which in multi-combat may not be the hitter. Resolve it properly, or record it as a known limit.
> 5. **Never pass `NONE` or a despawned NPC's index to `stillFighting`.** It revives a lapsed target however long ago it expired.
> 6. **Linger 0 still blinks** on ticks where interacting is not an NPC (`raw.log`: 97 of 127 changes). *Resolved in Task 5:* keep `min = 0`, per the spec's 0–60 s. A floor of 1 s wouldn't fix it: measured on `raw.log`, 119 of 133 evidence-free mid-fight stretches still blank at 1 s, 0 at 10 s. Linger 0 is **defined** as "show only on ticks with evidence of combat". The hitsplat path must refresh as well as interacting.
> 10. **Clamp config values in code. `@Range` is enforced only by the settings spinner.** In client 1.13.1 `ConfigPanel` is the only consumer of `getRange()`, so a profile can hold any int.
>    - **Task 6:** clamp `backgroundOpacity` to 0..100 before `Math.round(p * 255 / 100f)`. Opacity 101 gives alpha 258, and `new Color` throws on every render.
>    - **Task 7:** clamp `lingerSeconds` to 0..60 before converting it to milliseconds.
> 7. **Spec rule "health ratio -1 → last known value, dimmed" is not built anywhere** (Task 3 review). `onGameTick` rebuilds from live `getHealthRatio()` each tick, and `TargetSnapshot` has no last-known field.
```java
package com.foe;

import com.google.inject.Provides;
import javax.inject.Inject;
import lombok.Getter;
import net.runelite.api.Actor;
import net.runelite.api.Client;
import net.runelite.api.NPC;
import net.runelite.api.NPCComposition;
import net.runelite.api.Player;
import net.runelite.api.events.ActorDeath;
import net.runelite.api.events.GameTick;
import net.runelite.api.events.HitsplatApplied;
import net.runelite.api.events.InteractingChanged;
import net.runelite.api.events.NpcDespawned;
import net.runelite.client.config.ConfigManager;
import net.runelite.client.eventbus.Subscribe;
import net.runelite.client.game.NPCManager;
import net.runelite.client.plugins.Plugin;
import net.runelite.client.plugins.PluginDescriptor;
import net.runelite.client.ui.overlay.OverlayManager;

@PluginDescriptor(
	name = "Foe",
	description = "Live HP, combat levels and elemental weakness of the monster you're fighting",
	tags = {"target", "monster", "npc", "weakness", "opponent", "hp", "combat"}
)
public class FoePlugin extends Plugin
{
	@Inject
	private Client client;
	@Inject
	private FoeConfig config;
	@Inject
	private NPCManager npcManager;
	@Inject
	private OverlayManager overlayManager;
	@Inject
	private FoeOverlay overlay;

	private TargetTracker tracker;
	private NPC targetNpc;

	@Getter
	private volatile TargetSnapshot snapshot;

	@Provides
	FoeConfig provideConfig(ConfigManager configManager)
	{
		return configManager.getConfig(FoeConfig.class);
	}

	@Override
	protected void startUp()
	{
		tracker = new TargetTracker();
		overlayManager.add(overlay);
	}

	@Override
	protected void shutDown()
	{
		overlayManager.remove(overlay);
		targetNpc = null;
		snapshot = null;
	}

	@Subscribe
	public void onInteractingChanged(InteractingChanged e)
	{
		if (e.getSource() == client.getLocalPlayer() && isCombatNpc(e.getTarget()))
		{
			NPC npc = (NPC) e.getTarget();
			tracker.playerAttacks(npc.getIndex(), now());
			targetNpc = npc;
		}
	}

	@Subscribe
	public void onHitsplatApplied(HitsplatApplied e)
	{
		Player me = client.getLocalPlayer();
		if (e.getActor() instanceof NPC && e.getHitsplat().isMine() && isCombatNpc(e.getActor()))
		{
			// covers ranged/magic, where interacting can drop between attacks
			NPC npc = (NPC) e.getActor();
			tracker.playerAttacks(npc.getIndex(), now());
			targetNpc = npc;
		}
		else if (e.getActor() == me)
		{
			for (NPC npc : client.getTopLevelWorldView().npcs())
			{
				if (npc.getInteracting() == me && isCombatNpc(npc))
				{
					tracker.hitBy(npc.getIndex(), now(), lingerMs());
					if (tracker.current(now(), lingerMs()) == npc.getIndex())
					{
						targetNpc = npc;
					}
					break;
				}
			}
		}
	}

	@Subscribe
	public void onNpcDespawned(NpcDespawned e)
	{
		forget(e.getNpc());
	}

	@Subscribe
	public void onActorDeath(ActorDeath e)
	{
		if (e.getActor() instanceof NPC)
		{
			forget((NPC) e.getActor());
		}
	}

	@Subscribe
	public void onGameTick(GameTick e)
	{
		Player me = client.getLocalPlayer();
		if (targetNpc != null && me != null && (me.getInteracting() == targetNpc || targetNpc.getInteracting() == me))
		{
			tracker.stillFighting(targetNpc.getIndex(), now());
		}
		if (targetNpc == null || tracker.current(now(), lingerMs()) != targetNpc.getIndex())
		{
			targetNpc = null;
			snapshot = null;
			return;
		}
		NPCComposition c = targetNpc.getTransformedComposition();
		if (c == null)
		{
			snapshot = null;
			return;
		}
		snapshot = SnapshotFactory.build(targetNpc.getName(), c.getCombatLevel(), c.getStats(),
			targetNpc.getHealthRatio(), targetNpc.getHealthScale(), npcManager.getHealth(targetNpc.getId()),
			null); // weakness: Task 8
	}

	private void forget(NPC npc)
	{
		tracker.gone(npc.getIndex());
		if (npc == targetNpc)
		{
			targetNpc = null;
			snapshot = null;
		}
	}

	/** Spec addendum 2: Talk-to also sets interacting, so only NPCs with a combat level count. */
	private static boolean isCombatNpc(Actor a)
	{
		if (!(a instanceof NPC))
		{
			return false;
		}
		NPCComposition c = ((NPC) a).getTransformedComposition();
		return c != null && c.getCombatLevel() > 0;
	}

	private long lingerMs()
	{
		return config.lingerSeconds() * 1000L;
	}

	private static long now()
	{
		return System.currentTimeMillis();
	}
}
```
